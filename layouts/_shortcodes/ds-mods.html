{{/* ds-mods shortcode：饥荒单机版模组列表 用法：{{< ds-mods >}} 或 {{< ds-mods keyword="BM" searchOnly="true" >}} 【性能说明】 - 模组数据 JSON 从 site.Data.ds_pan 构建后，通过 Hugo ExecuteAsTemplate 生成独立静态资源 /ds-mods.json （由下面的 dataAsset 资源引用发布到站点，避免每个页面内嵌 20KB+ JSON） - 列表脚本 / 样式已抽成单独 ds-mods.js / ds-mods.css（assets/ 下），通过 pipeline fingerprint + 缓存 不再在每个 shortcode 页面重复内联 500+ 行模板代码 */}} {{- $panData := index site.Data "ds_pan" -}} {{- $currentDate := now.Format "20060102" -}} {{/* 命名参数：预填关键词 & 是否只显示搜索结果 */}} {{- $searchKeyword := .Get "keyword" | default "" -}} {{- $searchOnly := .Get "searchOnly" | default "false" -}} {{/* ========== 1. 构建标准化模组切片（BM000-BM999） ========== */}} {{- $bmItems := slice -}} {{- $bmCount := 0 -}} {{- range $id, $itemData := $panData -}} {{- if (findRE `^BM\d{3}$` (upper $id)) -}} {{- $bmCount = add $bmCount 1 -}} {{- $name := index (where $itemData "name" "!=" nil) 0 "name" | default "未知" -}} {{- $size := index (where $itemData "size" "!=" nil) 0 "size" -}} {{- $baiduUrl := index (where $itemData "url1" "ne" nil) 0 "url1" -}} {{- $quarkUrl := index (where $itemData "url3" "ne" nil) 0 "url3" -}} {{- $bmItems = $bmItems | append (dict "id" $id "name" $name "size" $size "baiduUrl" $baiduUrl "quarkUrl" $quarkUrl ) -}} {{- end -}} {{- end -}} {{- $sortedItems := sort $bmItems "id" "asc" -}} {{- $modsJson := $sortedItems | jsonify -}} {{/* ========== 2. 资源发布：把 JSON / CSS / JS 通过 Hugo pipeline 变成独立带指纹文件 ========== */}} {{/* JSON：通过 FromString + ExecuteAsTemplate（带日期后缀避免冲突） */}} {{- $dataAsset := $modsJson | resources.FromString (printf "data/ds-mods-%s.json" $currentDate) -}} {{- $dataAsset = $dataAsset | resources.Minify | resources.Fingerprint "sha256" -}} {{- /* CSS / JS：直接从 assets 读取再压缩 + 指纹 */ -}} {{- $cssAsset := resources.Get "styles/ds-mods.css" | resources.Minify | resources.Fingerprint "sha256" -}} {{- $jsAsset := resources.Get "js/ds-mods.js" | resources.Minify | resources.Fingerprint "sha256" -}} {{/* ========== 3. 输出：HTML 骨架 + 外链资源 + 短数据桥（
