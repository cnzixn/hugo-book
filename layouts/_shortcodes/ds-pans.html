{{/* 网盘下载短代码 用法: {{< pans "KEY_NAME" >}} 数据来源: data/ds-pan.yml 网盘总开关：hugo.yaml → params.panBaiduEnabled / panXunleiEnabled / panQuarkEnabled false 时对应网盘标签页不渲染；三个都关闭时只显示一条提示。 链接规范化：只保留「/s/」之后的分享段，用 params.panHosts 里的官方域名重建短链 （见 layouts/_partials/docs/pan-url.html），页面上显示/打开的都是重建后的链接。 */}} {{- $name := .Get 0 -}} {{- $panSw := partial "docs/pan-switches.html" . -}} {{- with index hugo.Data.ds_pan $name -}} {{- /* data 记录形如「单键 dict 的数组」：[{name: …}, {url1: …}, …]， 缺字段（如整条没有 url2）时 where 过滤后为空 → nil，交给 pan-url.html 按「无链接」处理 */ -}} {{- $baiduUrl := "" -}} {{- $xunleiUrl := "" -}} {{- $quarkUrl := "" -}} {{- range $k, $raw := (dict "baidu" (index (where . "url1" "ne" nil) 0 "url1") "xunlei" (index (where . "url2" "ne" nil) 0 "url2") "quark" (index (where . "url3" "ne" nil) 0 "url3")) -}} {{- /* 只保留「/s/」之后的分享段，用官方域名重建规范短链（前缀由 pan-url.html 统一丢弃） */ -}} {{- $pan := partial "docs/pan-url.html" (dict "url" $raw "site" $.Site) -}} {{- $url := "" -}} {{- if and $pan.ok (index $panSw $k) -}}{{- $url = $pan.url -}}{{- end -}} {{- if eq $k "baidu" -}}{{- $baiduUrl = $url -}} {{- else if eq $k "xunlei" -}}{{- $xunleiUrl = $url -}} {{- else -}}{{- $quarkUrl = $url -}}{{- end -}} {{- end -}} {{- $firstDisk := "baidu" -}} {{- if not $baiduUrl -}}{{- $firstDisk = "xunlei" -}}{{- end -}} {{- if not $baiduUrl -}}{{- if not $xunleiUrl -}}{{- $firstDisk = "quark" -}}{{- end -}}{{- end -}} {{- $itemName := index (where . "name" "ne" nil) 0 "name" -}} {{- $itemSize := index (where . "size" "ne" nil) 0 "size" -}} {{- /* size 为 0 / 0B（网盘里没量到大小）时不显示，避免出现「(0)」 */ -}} {{- if in (slice "0" "0B" "0b" "0.0" "0KB" "0.00B") (printf "%v" $itemSize) -}}{{- $itemSize = "" -}}{{- end -}} {{- $sizeHtml := "" -}} {{- if $itemSize -}}{{- $sizeHtml = printf " (%v)" $itemSize -}}{{- end -}} {{- if or $baiduUrl $xunleiUrl $quarkUrl -}}
{{- if $baiduUrl -}}
{{ $itemName }}{{ $sizeHtml | safeHTML }}
{{ $baiduUrl }}
{{- end -}} {{- if $xunleiUrl -}}
{{ $itemName }}{{ $sizeHtml | safeHTML }}
{{ $xunleiUrl }}
{{- end -}} {{- if $quarkUrl -}}
{{ $itemName }}{{ $sizeHtml | safeHTML }}
{{ $quarkUrl }}
{{- end -}}
{{- else -}}
{{ $itemName }}{{ $sizeHtml | safeHTML }}
暂无可用网盘下载
{{- end -}} {{- else -}}
网盘文件「{{ $name }}」分享链接已过期...
{{- end -}}
