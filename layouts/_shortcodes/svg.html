{{/* svg shortcode（PaperMod 风格）：站内图标查表输出 数据源：data/bs_icons.yml（图标名 → 完整 svg 标签），模板里读 hugo.Data.bs_icons 用法（位置参数 / 命名参数二选一，勿混用）： {{< svg "bs/envelope" >}} {{< svg name="bs/chat" size="1.5em" >}} 维护：新图标往 data/bs_icons.yml 追加一条 */}} {{- $icons := hugo.Data.bs_icons | default dict -}} {{- $name := strings.TrimPrefix "bs/" (.Get "name" | default (.Get 0)) -}} {{- $size := .Get "size" | default "1em" -}} {{- $svg := index $icons $name -}} {{- if not $svg -}} {{- /* 兼容不带 bi- 前缀的调用：globe2 → bi-globe2 */ -}} {{- $svg = index $icons (printf "bi-%s" $name) -}} {{- end -}} {{- if $svg -}} {{ $svg | safeHTML }} {{- else -}} {{- warnf "svg shortcode: 图标 %q 不在数据表里，渲染为占位符 ?" $name -}} ? {{- end -}}
