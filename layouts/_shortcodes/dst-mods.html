$panData -}} {{- if (findRE `^WS\d{6,20}$` (upper $id)) -}} {{- $wsCount = add $wsCount 1 -}} {{- $name := index (where $itemData "name" "!=" nil) 0 "name" | default "未知" -}} {{- $size := index (where if not $panSw.baidu -}}{{- $baiduUrl = "" -}}{{- end -}} {{- if not $panSw.xunlei -}}{{- $xunleiUrl = "" -}}{{- end -}} {{- if not $panSw.quark -}}{{- $quarkUrl = "" -}}{{- end -}} {{- $hasBaidu := gt "direct" -}} {{- /* 直链模式才把真实 url 写进 JSON（原代码保留，切回 direct 即恢复） */ -}} {{- if not (findRE `^https://pan\.baidu\.com/s/` $baiduUrl) -}}{{- $baiduUrl = "" -}}{{- end -}} {{- if not (findRE (dict "baiduUrl" $baiduUrl "xunleiUrl" $xunleiUrl "quarkUrl" $quarkUrl) -}} {{- end -}} {{- $wsItems = $wsItems | append $item -}} {{- end -}} {{- end -}} {{- $sortedItems := sort $wsItems "id" "asc" -}} {{- $modsJson := $sortedItems | jsonify -}} {{/* ========== 2. 资源发布（复用 ds-mods 的 CSS/JS 管道，JSON 单独命名 dst-） ========== */}} {{- $dataAsset := $modsJson | resources.FromString (printf
